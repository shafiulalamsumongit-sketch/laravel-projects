#Automatically Hydrating Parent Models on Children
class Post extends Model{
    public function comments(): HasMany{
        return $this->hasMany(Comment::class)->chaperone();
    }
}
$posts = Post::with(['comments' => fn ($comments) => $comments->chaperone(),])->get();

One to Many (Inverse) -> Belongs To
class Comment extends Model{
    public function post(): BelongsTo{
        return $this->belongsTo(Post::class);
		  return $this->belongsTo(Post::class, 'foreign_key');
		   return $this->belongsTo(Post::class, 'foreign_key', 'owner_key');
    }
}
$comment = Comment::find(1);
return $comment->post->title;




Default Models

The belongsTo, hasOne, hasOneThrough, and morphOne relationships allow you to define a default model that will be returned if the given relationship is null

/*** Get the author of the post.*/
public function user(): BelongsTo{
    return $this->belongsTo(User::class)->withDefault();
}


/*** Get the author of the post.*/
public function user(): BelongsTo{
    return $this->belongsTo(User::class)->withDefault([
        'name' => 'Guest Author',
    ]);
}

Has One of Many

latestOfMany in laravel
The latestOfMany method in Laravel's Eloquent ORM is used to retrieve the single latest record from a hasMany relationship, effectively turning it into a hasOne relationship for that specific latest record. This is particularly useful when you need to access the most recent related model without loading the entire collection of related models and then manually selecting the latest one.


class User extends Model
{
    public function latestLogin(): HasOne
    {
        return $this->hasOne(Login::class)->latestOfMany();
    }
}


$user = User::find(1);
$latestLogin = $user->latestLogin; // This will return a single Login model or null
